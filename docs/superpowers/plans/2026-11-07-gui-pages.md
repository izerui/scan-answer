```python
import threading
import time

from scan_answer.source.base import SheetImages
from scan_answer.source.control import ControlledSource


class Counting:
    def __init__(self, n):
        self.n = n
        self.calls = 0

    def sheets(self):
        self.calls += 1
        for i in range(self.n):
            yield SheetImages(source=str(i), names=[], dpis=[])


def test_pause_resume_continues_from_same_position():
    inner = Counting(5)
    src = ControlledSource(inner)
    got = []

    def consume():
        for item in src.sheets():
            got.append(item.source)
            if item.source == "1":
                src.pause()

    t = threading.Thread(target=consume)
    t.start()
    time.sleep(0.3)
    assert got == ["0", "1"] and src.paused
    src.resume()
    t.join(2)
    assert got == ["0", "1", "2", "3", "4"] and inner.calls == 1 and src.taken == 5


def test_stop_while_paused_ends_iteration():
    src = ControlledSource(Counting(5))
    it = src.sheets()
    assert next(it).source == "0"
    src.pause()
    threading.Timer(0.2, src.stop).start()
    assert list(it) == [] and src.stopped and not src.paused
```

`src/scan_answer/pipeline/runner.py`（整文件替换）:
```python
"""流水线：图像源 → 保存原图 → 识别进程池 → 记录结果 → 发事件；结束批次时按最终阈值重新判定。

扫描与识别并行：主线程从图像源取卡、提交给进程池，同时收集已完成的结果；
在途任务数不超过 workers*2，防止扫描远快于识别时内存堆积。workers=0 时在当前进程内顺序执行（测试和调试用）。
所有事件都在调用 run()/rescan()/finalize() 的线程里发出。
"""
from __future__ import annotations

import multiprocessing
import threading
from concurrent.futures import FIRST_COMPLETED, Executor, Future, ProcessPoolExecutor, wait
from concurrent.futures.process import BrokenProcessPool
from typing import Callable

from ..source.base import DeviceEvent, ImageSource, SheetImages
from .events import DeviceNotice, FinalizedEvent, SheetEvent, SheetQueued, ThresholdEvent, VoidEvent
from .session import BatchSession
from .worker import FinalizeJob, SheetJob, SheetOutcome, finalize_sheet, process_sheet

Listener = Callable[[object], None]


class _InlineExecutor(Executor):
    """在当前线程立即执行的执行器，接口与进程池一致。"""

    def submit(self, fn, /, *args, **kwargs) -> Future:
        f: Future = Future()
        try:
            f.set_result(fn(*args, **kwargs))
        except BaseException as e:  # noqa: BLE001 — 交给调用方的 result() 抛出
            f.set_exception(e)
        return f


class BatchPipeline:
    def __init__(self, session: BatchSession, *, workers: int = 0, listener: Listener | None = None,
                 on_result: Callable[[str], None] | None = None):
        """on_result：每张卡成功写出 result.json 后回调其绝对路径（用于投递手写转写）。"""
        self.session = session
        self.workers = workers
        self.listener = listener or (lambda e: None)
        self.on_result = on_result
        self._gen = 0  # 执行器代数：每重建一次加 1
        self._executor: Executor = self._make_executor()

    # ---------- 生命周期 ----------
    def close(self) -> None:
        self._executor.shutdown(wait=True)

    def __enter__(self) -> "BatchPipeline":
        return self

    def __exit__(self, *exc) -> None:
        self.close()

    # ---------- 内部 ----------
    def _make_executor(self) -> Executor:
        if self.workers > 0:
            return ProcessPoolExecutor(self.workers, mp_context=multiprocessing.get_context("spawn"))
        return _InlineExecutor()

    def _rebuild(self, gen: int) -> None:
        """进程池损坏（某个识别进程异常退出）时重建；同一代只重建一次。"""
        if gen != self._gen:
            return
        old = self._executor
        self._executor = self._make_executor()
        self._gen += 1
        try:
            old.shutdown(wait=False, cancel_futures=True)
        except Exception:  # noqa: BLE001 — 已损坏的池关闭失败不影响后续
            pass

    def _submit(self, fn, job) -> tuple[Future, int]:
        """提交任务，返回 (future, 提交时的执行器代数)；进程池已损坏时重建后再提交一次。"""
        err: BaseException | None = None
        for _ in range(2):
            gen = self._gen
            try:
                return self._executor.submit(fn, job), gen
            except BrokenProcessPool as e:
                err = e
                self._rebuild(gen)
        f: Future = Future()
        f.set_exception(err)
        return f, self._gen

    def _result(self, f: Future, gen: int, fn, job):
        """取结果；进程池损坏导致失败时重建并重提交一次，再失败则抛出。"""
        try:
            return f.result()
        except BrokenProcessPool:
            self._rebuild(gen)
        f2, gen2 = self._submit(fn, job)
        try:
            return f2.result()
        except BrokenProcessPool:
            self._rebuild(gen2)  # 保证后续任务有可用的进程池
            raise

    def _outcome(self, f: Future, gen: int, job: SheetJob) -> SheetOutcome:
        """安全地取结果；worker 异常时转为物理异常记录，不中断批次。"""
        try:
            return self._result(f, gen, process_sheet, job)
        except Exception as e:  # noqa: BLE001 — worker 崩溃不应中断整批
            return SheetOutcome(job.seq, False, physical="ALIGN_FAILED",
                                message=f"内部错误：{type(e).__name__}: {e}")

    def _job(self, seq: int, sheet: SheetImages) -> SheetJob:
        s = self.session
        raw = s.save_raw(seq, sheet)
        m = s.manifest
        return SheetJob(seq=seq, raw_paths=[str(p) for p in raw], dpis=list(sheet.dpis),
                        mark=s.thresholds.mark, uncertain=s.thresholds.uncertain,
                        batch_dir=str(s.batch_dir), template_dir=str(s.template_dir), batch_id=s.batch_id,
                        exam=m["exam"], scanner=m["scanner"], ocr=m["ocr"])

    def _collect(self, out: SheetOutcome, rescan: bool = False) -> SheetEvent:
        refit = self.session.record(out)
        item = self.session.sheets[out.seq]
        ev = SheetEvent(out.seq, item["source"], out.ok, out.physical, out.message, out.exam_no,
                        out.warnings, rescan, str(self.session.batch_dir / out.result) if out.ok else None)
        self.listener(ev)
        if out.ok and self.on_result:
            self.on_result(str(self.session.batch_dir / out.result))
        if refit:
            self.listener(ThresholdEvent(refit.mark, refit.uncertain, len(self.session.fills)))
        return ev

    # ---------- 对外 ----------
    def run(self, source: ImageSource, stop: threading.Event | None = None) -> None:
        """处理图像源中的全部卡，直到取完或 stop 被置位。返回前等待所有在途任务完成。"""
        pending: dict[Future, tuple[int, SheetJob]] = {}
        window = max(1, self.workers * 2)
        try:
            for item in source.sheets():
                if stop is not None and stop.is_set():
                    break
                if isinstance(item, DeviceEvent):
                    self.session.add_event(item.kind, item.message)
                    self.listener(DeviceNotice(item.kind, item.message, self.session.next_seq() - 1))
                    continue
                seq = self.session.next_seq()
                self.session.register(seq, item)
                self.listener(SheetQueued(seq, item.source))
                job = self._job(seq, item)
                f, gen = self._submit(process_sheet, job)
                pending[f] = (gen, job)
                if len(pending) >= window:
                    done, _ = wait(pending, return_when=FIRST_COMPLETED)
                    for f in done:
                        gen, job = pending.pop(f)
                        self._collect(self._outcome(f, gen, job))
            for f, (gen, job) in list(pending.items()):
                self._collect(self._outcome(f, gen, job))
        finally:
            self.session.save(force=True)

    def rescan(self, seq: int, sheet: SheetImages) -> SheetEvent:
        """用新图像替换某个序号的卡（物理异常重扫）；序号不变。"""
        self.session.prepare_rescan(seq)
        self.session.register(seq, sheet)
        self.listener(SheetQueued(seq, sheet.source))
        job = self._job(seq, sheet)
        f, gen = self._submit(process_sheet, job)
        ev = self._collect(self._outcome(f, gen, job), rescan=True)
        self.session.save(force=True)
        return ev

    def void(self, seq: int) -> None:
        self.session.void(seq)
        self.listener(VoidEvent(seq))

    def finalize(self) -> FinalizedEvent:
        """结束批次：用整批涂黑度拟合最终阈值，逐张重新判定；有变化的卡重写 result.json。"""
        s = self.session
        th = s.final_thresholds()
        jobs = [FinalizeJob(seq, th.mark, th.uncertain, str(s.batch_dir), str(s.template_dir))
                for seq, item in sorted(s.sheets.items()) if item.get("status") == "ok"]
        changed: list[int] = []
        futures = [(j, *self._submit(finalize_sheet, j)) for j in jobs]
        for j, f, gen in futures:
            try:
                r = self._result(f, gen, finalize_sheet, j)
            except Exception as e:  # noqa: BLE001 — 单张失败不中断整批结束
                s.add_event("error", f"重新判定失败 #{j.seq}: {type(e).__name__}: {e}")
                continue
            with s.lock:
                item = s.sheets[r.seq]
                item["warnings"] = r.warnings
                item["exam_no"] = r.exam_no
            if r.changed:
                changed.append(r.seq)
        s.mark_finalized(th)
        ev = FinalizedEvent(th.mark, th.uncertain, len(changed), changed)
        self.listener(ev)
        return ev
```

```python
"""可暂停、可停止的图像源包装。

界面的“暂停 / 继续”要从上次的位置接着取卡，所以包装器只创建一次底层迭代器，
暂停时阻塞在取下一张之前，停止时结束迭代。供流水线所在的后台线程使用，暂停/继续/停止可在任意线程调用。
"""
from __future__ import annotations

import threading
from typing import Iterator

from .base import ImageSource, SourceItem


class ControlledSource:
    def __init__(self, inner: ImageSource):
        self._it: Iterator[SourceItem] = iter(inner.sheets())
        self._running = threading.Event()
        self._running.set()
        self._stopped = threading.Event()
        self.taken = 0  # 已交给流水线的条目数

    # ---------- 控制（任意线程） ----------
    def pause(self) -> None:
        self._running.clear()

    def resume(self) -> None:
        self._running.set()

    def stop(self) -> None:
        self._stopped.set()
        self._running.set()  # 唤醒可能在等待的迭代

    @property
    def paused(self) -> bool:
        return not self._running.is_set() and not self._stopped.is_set()

    @property
    def stopped(self) -> bool:
        return self._stopped.is_set()

    # ---------- 迭代（流水线线程） ----------
    def sheets(self) -> Iterator[SourceItem]:
        while True:
            self._running.wait()
            if self._stopped.is_set():
                return
            try:
                item = next(self._it)
            except StopIteration:
                return
            self.taken += 1
            yield item
```

```python
目录约定：

@dataclass
class TemplateInfo:
    name: str
    version: int
    directory: Path
    paper: tuple[float, float]
    faces: int
    questions: int  # 客观题 + 主观题数量


def _template_info(d: Path) -> TemplateInfo | None:
    p = d / "template.json"
    if not p.exists():
        return None
    try:
        t = json.loads(p.read_text(encoding="utf-8"))
        n_obj = sum(len(b["grid"]["cells"]) for b in t.get("objective", []))
        return TemplateInfo(t["name"], int(t["version"]), d, tuple(t["paper"]), len(t["faces"]),

def list_templates(dirs: list[str]) -> list[TemplateInfo]:
    """每个目录本身是模板，或其子目录是模板；按名称排序、去重。"""
    seen, out = set(), []
    for raw in dirs:
        d = Path(raw)
        candidates = [d] + (sorted(x for x in d.iterdir() if x.is_dir()) if d.is_dir() else [])
        for c in candidates:
            info = _template_info(c)
            if info and info.directory.resolve() not in seen:
                seen.add(info.directory.resolve())
                out.append(info)
    return sorted(out, key=lambda t: (t.name, t.version))

def create_exam(output_root: str | Path, info: ExamInfo, template_dir: str | Path) -> Exam:
    root = Path(output_root)
    stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    safe = "".join(ch for ch in f"{info.name}_{info.subject}" if ch not in '\\/:*?"<>|').strip("_ ") or "考试"

    n = 1
    while d.exists():
        n += 1
        d = root / f"{safe}_{stamp}_{n}"
    d.mkdir(parents=True)
    created = datetime.now().astimezone().isoformat(timespec="seconds")
    write_json_atomic(d / "exam.json", {"info": asdict(info), "template_dir": str(Path(template_dir)),
                                        "created": created})

        out.append(BatchSummary(m.get("batch_id", d.name), d, m.get("created", ""), m.get("counts") or {},
```

```python
import json

from pathlib import Path

class StudentRow:

    m = json.loads((b / "manifest.json").read_text(encoding="utf-8"))
    rows = []
    for s in m.get("sheets", []):
        rows.append(StudentRow(int(s["seq"]), s.get("source", ""), s.get("status", ""), s.get("exam_no"),
                               int(s.get("warnings", 0)), b / s["result"] if s.get("result") else None,
                               s.get("physical"), s.get("message", "")))
    return rows


def load_batch_template(batch_dir: str | Path) -> Template:
    """批次内的模板快照（新批次在 template/，旧批次在根目录 template.json）。"""
    b = Path(batch_dir)
    p = b / "template" / "template.json"
    if not p.exists():
        p = b / "template.json"
    return template_from_dict(json.loads(p.read_text(encoding="utf-8")))


@dataclass
class OverlayMM:
    rect: Rect  # mm
    kind: str  # answer / bad / exam / exam_bad
    label: str = ""


def _union(rects: list[Rect]) -> Rect:
    x0 = min(r[0] for r in rects)
    y0 = min(r[1] for r in rects)
    x1 = max(r[0] + r[2] for r in rects)
    y1 = max(r[1] + r[3] for r in rects)
    return (x0, y0, x1 - x0, y1 - y0)


def face_overlays(tpl: Template, result: dict, face: str) -> list[OverlayMM]:
    """某一面的叠层：识别到的选项（answer）、非 ok 的题整行（bad，带题号与状态）、考号所涂格（exam / exam_bad）。"""
    out: list[OverlayMM] = []
    status_text = {"blank": "未答", "multiple": "多涂", "uncertain": "存疑"}
    objective = {o["no"]: o for o in result.get("objective", [])}
    for blk in tpl.objective:
        if blk.grid.face != face:
            continue
        for k, cells in enumerate(blk.grid.cells):
            o = objective.get(str(blk.start + k))
            if not o:
                continue
            for i, lab in enumerate(blk.grid.labels):
                if lab in o.get("answer", ""):
                    out.append(OverlayMM(cells[i], "answer" if o["status"] == "ok" else "bad"))
            if o["status"] != "ok":
                out.append(OverlayMM(_union(cells), "bad", f"{o['no']} {status_text.get(o['status'], o['status'])}"))
    exam = result.get("student", {}).get("exam_no")
    if tpl.exam_no and tpl.exam_no.grid.face == face and exam:
        bad_pos = {d["pos"] for d in exam.get("digits", [])}
        for pos, (ch, cells) in enumerate(zip(exam.get("value", ""), tpl.exam_no.grid.cells), start=1):
            if pos in bad_pos:
                out.append(OverlayMM(_union(cells), "exam_bad", f"第{pos}位"))
            elif ch in tpl.exam_no.grid.labels:
                out.append(OverlayMM(cells[tpl.exam_no.grid.labels.index(ch)], "exam"))
    return out
```

`src/scan_answer/ui/widgets/mosaic.py`:
```python
"""批次拼图：每格一张纸、每行 50 张，形状 + 颜色双编码。

成功 = 浅绿方块；数据警告 = 琥珀扇形；物理异常 = 红圆；识别中 = 蓝方块；作废 = 灰色叉；待扫 = 空框。
悬停显示序号与原因，点击发 cellClicked(seq)。批次超过 1000 张时默认只显示最近 10 行，可展开。
"""
from __future__ import annotations

import math
from dataclasses import dataclass

from PySide6.QtCore import QPointF, QRectF, QSize, Qt, Signal
from PySide6.QtGui import QPainter, QPainterPath, QPen
from PySide6.QtWidgets import QToolTip, QWidget

from .. import theme

PER_ROW = 50
RULER_W = 44
GAP = 2
COLLAPSE_AFTER = 1000
COLLAPSED_ROWS = 10

QUEUED, OK, WARN, ERR, VOID = "queued", "ok", "warn", "err", "void"
LABELS = {QUEUED: "识别中", OK: "成功", WARN: "数据警告", ERR: "物理异常", VOID: "已作废", None: "待扫"}


@dataclass
class Cell:
    state: str
    note: str = ""


class BatchMosaic(QWidget):
    cellClicked = Signal(int)

    def __init__(self, parent: QWidget | None = None):
        super().__init__(parent)
        self.cells: dict[int, Cell] = {}
        self.planned = 0  # 计划张数（未知时为 0，拼图随已登记张数增长）
        self.current: int | None = None
        self.expanded = False
        self.setMouseTracking(True)
        self.setMinimumHeight(60)

    # ---------- 数据 ----------
    def reset(self, planned: int = 0) -> None:
        self.cells.clear()
        self.planned = planned
        self.current = None
        self.updateGeometry()
        self.update()

    def set_cell(self, seq: int, state: str, note: str = "") -> None:
        self.cells[seq] = Cell(state, note)
        if state == QUEUED:
            self.current = seq
        self.updateGeometry()
        self.update()

    def count(self, state: str) -> int:
        return sum(1 for c in self.cells.values() if c.state == state)

    # ---------- 几何 ----------
    def total_slots(self) -> int:
        n = max(self.planned, max(self.cells, default=0))
        return max(PER_ROW, math.ceil(n / PER_ROW) * PER_ROW)

    def first_row(self) -> int:
        rows = self.total_slots() // PER_ROW
        if self.expanded or self.total_slots() <= COLLAPSE_AFTER:
            return 0
        return max(0, rows - COLLAPSED_ROWS)

    def visible_rows(self) -> int:
        return self.total_slots() // PER_ROW - self.first_row()

    def cell_size(self) -> float:
        return max(6.0, (self.width() - RULER_W - GAP * (PER_ROW - 1)) / PER_ROW)

    def _row_y(self, r: int, size: float) -> float:
        """第 r 个可见行的顶端；每满 100 张（两行）多留 4 像素。"""
        first = self.first_row()
        breaks = sum(1 for i in range(1, r + 1) if (i + first) % 2 == 0)
        return r * (size + GAP) + breaks * 4

    def sizeHint(self) -> QSize:
        return QSize(760, self.heightForWidth(760))

    def hasHeightForWidth(self) -> bool:
        return True

    def heightForWidth(self, w: int) -> int:
        size = max(6.0, (w - RULER_W - GAP * (PER_ROW - 1)) / PER_ROW)
        rows = self.visible_rows()
        return int(self._row_y(rows - 1, size) + size + 2)

    def cell_rect(self, seq: int) -> QRectF | None:
        idx = seq - 1
        r = idx // PER_ROW - self.first_row()
        if r < 0 or r >= self.visible_rows():
            return None
        s = self.cell_size()
        c = idx % PER_ROW
        return QRectF(RULER_W + c * (s + GAP), self._row_y(r, s), s, s)

    def seq_at(self, pos: QPointF) -> int | None:
        s = self.cell_size()
        if pos.x() < RULER_W:
            return None
        c = int((pos.x() - RULER_W) // (s + GAP))
        if c >= PER_ROW:
            return None
        for r in range(self.visible_rows()):
            y = self._row_y(r, s)
            if y - 1 <= pos.y() <= y + s + 1:
                seq = (r + self.first_row()) * PER_ROW + c + 1
                rect = self.cell_rect(seq)
                return seq if rect and rect.adjusted(-1, -1, 1, 1).contains(pos) else None
        return None

    # ---------- 绘制 ----------
    def paintEvent(self, _e) -> None:
        p = QPainter(self)
        p.setRenderHint(QPainter.RenderHint.Antialiasing)
        p.setFont(theme.ui_font(12))
        first = self.first_row()
        for r in range(self.visible_rows()):
            row_first = (r + first) * PER_ROW + 1
            rect0 = self.cell_rect(row_first)
            p.setPen(theme.qcolor(theme.INK3))
            p.drawText(QRectF(0, rect0.top() - 2, RULER_W - 6, rect0.height() + 4),
                       Qt.AlignmentFlag.AlignLeft | Qt.AlignmentFlag.AlignVCenter, str(row_first))
            for c in range(PER_ROW):
                seq = row_first + c
                self._draw_cell(p, self.cell_rect(seq), self.cells.get(seq), seq == self.current)
```

```python
"""扫描控制器：把流水线放到后台线程，界面只发命令、收信号。

BatchSession / BatchPipeline 不是线程安全的，所以界面线程从不直接调用它们：
所有操作（开始、重扫、作废、结束批次）进入命令队列，由同一个后台线程串行执行；
暂停 / 继续 / 停止通过 ControlledSource 生效（可从任意线程调用）。
流水线事件和转写进度在后台线程发出，经 Qt 信号（跨线程自动排队）送到界面线程。
"""
from __future__ import annotations

import queue
import threading
from pathlib import Path

from PySide6.QtCore import QObject, Signal

from ..app.library import Exam
from ..batch import ocr_progress_listener
from ..ocr.config import OcrConfig
from ..ocr.queue import OcrQueue
from ..pipeline.events import (DeviceNotice, FinalizedEvent, OcrProgress, SheetEvent, SheetQueued, ThresholdEvent,
                               VoidEvent)
from ..pipeline.runner import BatchPipeline
from ..pipeline.session import BatchSession
from ..source.base import ImageSource, SheetImages
from ..source.control import ControlledSource
from ..source.folder import read_dpi

IDLE, RUNNING, PAUSED, FINISHING, FINALIZED = "idle", "running", "paused", "finishing", "finalized"


class ScanController(QObject):
    sheetQueued = Signal(int, str)
    sheetDone = Signal(object)  # SheetEvent
    thresholdChanged = Signal(object)  # ThresholdEvent
    deviceNotice = Signal(object)  # DeviceNotice
    voided = Signal(int)
    finalized = Signal(object)  # FinalizedEvent
    ocrProgress = Signal(object)  # OcrProgress
    stateChanged = Signal(str)
    errorOccurred = Signal(str)

    def __init__(self, workers: int = 0, parent: QObject | None = None):
        super().__init__(parent)
        self.workers = workers
        self.session: BatchSession | None = None
        self._pipeline: BatchPipeline | None = None
        self._ocr: OcrQueue | None = None
        self._source: ControlledSource | None = None
        self._cmds: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._loop, name="scan-pipeline", daemon=True)
        self._thread.start()
        self.state = IDLE

    # ---------- 事件转信号（后台线程） ----------
    def _on_event(self, e: object) -> None:
        if isinstance(e, SheetQueued):
            self.sheetQueued.emit(e.seq, e.source)
        elif isinstance(e, SheetEvent):
            self.sheetDone.emit(e)
        elif isinstance(e, ThresholdEvent):
            self.thresholdChanged.emit(e)
        elif isinstance(e, DeviceNotice):
            self.deviceNotice.emit(e)
        elif isinstance(e, VoidEvent):
            self.voided.emit(e.seq)
        elif isinstance(e, FinalizedEvent):
            self.finalized.emit(e)
        elif isinstance(e, OcrProgress):
            self.ocrProgress.emit(e)

    def _set_state(self, s: str) -> None:
        self.state = s
        self.stateChanged.emit(s)

    # ---------- 后台线程 ----------
    def _loop(self) -> None:
        while True:
            cmd, args = self._cmds.get()
            if cmd == "quit":
                return
            try:
                getattr(self, "_do_" + cmd)(*args)
            except Exception as e:  # noqa: BLE001 — 报给界面，线程继续服务后续命令
                self.errorOccurred.emit(f"{type(e).__name__}: {e}")
                if cmd == "run":
                    self._set_state(IDLE)

    def _do_run(self, source: ControlledSource) -> None:
        # 界面可能在命令排队期间就点了暂停：以图像源的实际状态为准
        self._set_state(PAUSED if source.paused else RUNNING)
        self._pipeline.run(source)
        self._source = None
        self._set_state(IDLE if not self.session.manifest["finalized"] else FINALIZED)

    def _do_rescan(self, seq: int, sheet: SheetImages) -> None:
        self._pipeline.rescan(seq, sheet)

    def _do_void(self, seq: int) -> None:
        self._pipeline.void(seq)

    def _do_finalize(self) -> None:
        self._set_state(FINISHING)
        self._pipeline.finalize()
        self._set_state(FINALIZED)

    def _do_close_batch(self, done: threading.Event) -> None:
        try:
            if self._pipeline:
                self._pipeline.close()
            if self._ocr:
                self._ocr.close()
        finally:
            self._pipeline = self._ocr = self.session = None
            done.set()

    # ---------- 对外（界面线程） ----------
    def _open(self, session: BatchSession, ocr_cfg: OcrConfig | None) -> BatchSession:
        self.close_batch()
        self.session = session
        self._ocr = OcrQueue(ocr_cfg, ocr_progress_listener(session, self._on_event)) if ocr_cfg else None
        self._pipeline = BatchPipeline(session, workers=self.workers, listener=self._on_event,
                                       on_result=self._ocr.enqueue if self._ocr else None)
        self._set_state(FINALIZED if session.manifest["finalized"] else IDLE)
        return session

    def new_batch(self, exam: Exam, ocr_cfg: OcrConfig | None = None, scanner: str = "folder") -> BatchSession:
        session = BatchSession.create(exam.directory, exam.template_dir, exam=exam.info.meta(), scanner=scanner,
                                      ocr=ocr_cfg is not None)
        return self._open(session, ocr_cfg)

    def open_batch(self, batch_dir: str | Path, ocr_cfg: OcrConfig | None = None) -> BatchSession:
        """继续已有批次（含程序崩溃后重开）；待转写项重新投递。"""
        session = self._open(BatchSession.open(batch_dir), ocr_cfg)
        if self._ocr:
            for item in session.sheets.values():
                if item.get("status") == "ok" and item.get("result"):
                    self._ocr.enqueue(session.batch_dir / item["result"])
        return session

    def start(self, source: ImageSource) -> None:
        if self.session is None:
            raise RuntimeError("请先新建或打开批次")
        if self._source is not None:
            raise RuntimeError("当前批次正在扫描")
        self._source = ControlledSource(source)
        self._cmds.put(("run", (self._source,)))

    def pause(self) -> None:
        if self._source:
            self._source.pause()
            self._set_state(PAUSED)

    def resume(self) -> None:
        if self._source:
            self._source.resume()
            self._set_state(RUNNING)

    def stop(self) -> None:
        if self._source:
            self._source.stop()

    def rescan(self, seq: int, files: list[str | Path]) -> None:
        paths = [Path(f) for f in files]
        sheet = SheetImages(source=paths[0].stem if paths else "重扫", names=[p.name for p in paths],
                            dpis=[read_dpi(p) for p in paths], paths=paths)
        self._cmds.put(("rescan", (seq, sheet)))

    def void(self, seq: int) -> None:
        self._cmds.put(("void", (seq,)))

    def finalize(self) -> None:
        self.stop()
        self._cmds.put(("finalize", ()))

    def close_batch(self, timeout: float = 60.0) -> None:
        """停止当前批次并释放进程池、转写线程（等待在途任务完成）。"""
        if self.session is None:
            return
        self.stop()
        done = threading.Event()
        self._cmds.put(("close_batch", (done,)))
        done.wait(timeout)
        self._set_state(IDLE)

    def shutdown(self) -> None:
        self.close_batch()
        self._cmds.put(("quit", ()))
        self._thread.join(timeout=10)

    def wait_idle(self, timeout: float = 60.0) -> bool:
        """等待已提交的命令全部执行完（测试与“结束批次”后的收尾用）。"""
        done = threading.Event()
        self._cmds.put(("noop", (done,)))
        return done.wait(timeout)

    def _do_noop(self, done: threading.Event) -> None:
        done.set()

    def wait_ocr(self) -> None:
        if self._ocr:
            self._ocr.wait()
```

实现要点：
- 命令队列 + 一个后台线程，保证 `BatchSession`/`BatchPipeline` 只在一个线程里被调用。
- `_do_run` 开始时以图像源的实际状态设置 `running/paused`：界面可能在命令排队期间就点了暂停（原型中实测到的竞态）。
- 信号在后台线程或转写线程中发出，接收方在界面线程，Qt 自动按队列连接投递。

````markdown
## 状态取值

| 字段 | 取值 |
|---|---|
| `status`（客观题、考号、考号逐位、条码） | `ok` / `blank` / `multiple` / `uncertain` / `partial` / `failed` |
| `ocr_status` | `none`（未开启转写）/ `pending`（等待转写）/ `done` / `failed` |
| `scan.faces.*.align_method` | `anchors` / `frames` / `features` |

## 物理异常代码（不生成 result.json，写入 errors/）

| 代码 | 含义 | 现场处理 |
|---|---|---|
| `ALIGN_FAILED` | 所有面都无法定位、图片文件无法读取，或识别时发生内部错误（说明以"内部错误："开头） | 检查折角、放反，重扫 |
| `FACE_MISMATCH` | 能定位但与模板正反面对不上（扫错卡、混入其他纸） | 核对后重扫或作废 |
| `BLANK_PAGE` | 有一面几乎全白 | 检查是否漏进纸、双张 |
| `MISSING_FACE` | 图像数量不足一张卡的面数 | 补扫 |
| `SKEW_TOO_LARGE` | 倾斜超过 8° | 摆正后重扫 |

| 代码 | 含义 | `field` |
|---|---|---|

| `EXAM_NO_BLANK` | 考号整体未填涂 | `student.exam_no` |
| `ABSENT_MARKED` | 缺考标记已填涂 | `student.absent` |

| 字段 | 说明 |
|---|---|
| `seq`、`batch_id` | 扫描序号、批次号 |
| `template` | `{"name", "version"}` |

| `objective[]` | `no`、`type`、`answer`（选中选项按顺序拼接）、`status`、`fill`（每个选项的涂黑度 0~1）、`image`（仅非 ok 时） |

| `omr_fields`、`choose_one` | 本期为 `{}`、`[]` |
| `manual` | 预留人工校对，本期 `null` |

## manifest.json

| 字段 | 说明 |
|---|---|
| `batch_id`、`exam`、`template`、`scanner`、`created` | 批次信息；`template.snapshot` 为快照路径 `template/template.json`（旧批次为 `template.json`） |
| `ocr` | 本批次是否开启手写转写 |
| `thresholds` | 当前判定阈值 `{"mark", "uncertain"}`：扫描中先用默认值，成功识别满 100 张后改用已扫部分的拟合值（之后每 50 张更新）；结束批次后为整批拟合的最终值 |
| `finalized` | 是否已结束批次。结束批次时用最终阈值逐张重新判定，判定有变化的卡会重写 `result.json`（保留已有转写结果）；未结束的批次中个别临界题的状态以后可能变化 |
| `counts` | `total`、`ok`、`physical`、`voided`、`warnings`（成功卡识别警告数之和，口径同 `sheets[].warnings`） |
| `ocr_progress` | 手写转写进度 `{"done", "failed", "total"}`，转写中约每 2 秒更新、转写结束时写入最终值；未开启转写的批次为 `null`。`scan-answer ocr` 命令运行后记录的是该次运行的进度 |
| `events[]` | 扫描仪事件：`time`、`kind`（`double_feed` / `paper_jam` / `no_paper` / `error`）、`message`、`after_seq`（事件发生时已分配的最大序号） |
| `sheets[]` | 每张卡：`seq`、`source`、`files`、`status`（`processing` / `ok` / `physical` / `voided`）；成功时有 `result`、`exam_no`、`warnings`（识别警告数，不含 `OCR_FAILED`；转写失败只在 `result.json` 的 `warnings` 与 `ocr_progress` 中体现）；物理异常时有 `physical`、`message`；作废时 `voided: true` |

### 重扫与作废

- 只能重扫 `status` 为 `physical`（物理异常）或 `voided`（已作废）的卡，序号不变；重扫成功的卡不能再重扫。
- 只能作废 `physical` 的卡。

### 程序崩溃后重开批次

每张卡开始识别前先把 `processing` 条目写入 manifest。重开批次时按磁盘内容对账：

- 新序号从 manifest 与 `raw/`、`state/`、`students/`、`errors/` 中出现过的最大序号之后继续，不会复用。
- manifest 中为 `processing` 的条目、以及磁盘上有但 manifest 中没有的序号：有 `state/{seq}.json` 则恢复为 `ok`（`exam_no`、`warnings` 取自 `result.json`）；否则有 `errors/{seq}_{代码}/error.json` 则恢复为对应的物理异常；都没有则记为 `ALIGN_FAILED`，`message` 为“识别中断（程序退出前未完成），请重扫”。
- manifest 中原本没有的序号，`source` 与 `files` 取自 `raw/` 中的原图文件名（去掉序号前缀），没有原图时 `source` 为“恢复”。

## 手写转写

- 识别时提供转写配置（`recognize --ocr-config`）则主观题与信息栏的 `ocr_status` 为 `pending`，后台转写完成后回写 `text`、`ocr_model`、`ocr_confidence`。
- 模型直接输出转写内容（数学公式为 LaTeX，反斜杠原样保留），最后单独一行写“置信度: 0.xx”；`ocr_confidence` 即取自该行（0~1，没有该行或数值越界时为 `null`），该行不计入 `text`。
- 配置文件（JSON）：`{"base_url": "https://.../v1", "api_key": "...", "model": "...", "concurrency": 4, "timeout_s": 60, "max_retries": 2}`，接口为 OpenAI Chat Completions 兼容格式。
- 事后补做或重试：`scan-answer ocr --batch <批次目录> --config <配置> [--enable] [--retry-failed]`；有失败项时退出码为 1。
- `scan-answer ocr` 只能对**已停止扫描**的批次运行：它会打开批次并做一次磁盘对账，若同时有程序在扫描同一批次，正在识别的卡会被误标为“识别中断”。
````
